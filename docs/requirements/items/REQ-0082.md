$$
\text{CalcDef} \le \text{Function}
$$

$$

$$

$$
\forall \mathbf{x} \in \text{dom}(f), \quad \Delta \text{State} \equiv \emptyset \wedge f(\mathbf{x}) = f(\mathbf{x})
